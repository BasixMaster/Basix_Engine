cmake_minimum_required(VERSION 3.24)
project(BasixEngine VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(basix_cage src/cage.cpp)
target_include_directories(basix_cage PUBLIC include)
if(MSVC)
 target_compile_options(basix_cage PRIVATE /W4 /permissive-)
else()
 target_compile_options(basix_cage PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
 add_executable(basix_cage_tests tests/cage_tests.cpp)
 target_link_libraries(basix_cage_tests PRIVATE basix_cage)
 add_test(NAME cage_correctness COMMAND basix_cage_tests)
endif()
option(BASIX_BUILD_GPU_SPIKE "Build NoGraphicsAPI Vulkan GPU experiments" OFF)
set(BASIX_NOGRAPHICSAPI_SOURCE "" CACHE PATH "Pinned, locally prepared NoGraphicsAPI source (see tools/prepare_nographicsapi.py)")
if(BASIX_BUILD_GPU_SPIKE)
 if(APPLE)
  message(FATAL_ERROR "The Basix GPU spike is Vulkan-only. Apple/Metal support is not implemented; the CPU target remains available.")
 endif()
 if(NOT BASIX_NOGRAPHICSAPI_SOURCE)
  message(FATAL_ERROR "Set BASIX_NOGRAPHICSAPI_SOURCE to a prepared pinned copy. No dependency is downloaded automatically.")
 endif()
 find_package(Vulkan 1.4.357 REQUIRED)
 find_program(BASIX_SLANGC slangc REQUIRED)
 execute_process(COMMAND ${BASIX_SLANGC} -version OUTPUT_VARIABLE slang_version ERROR_VARIABLE slang_error RESULT_VARIABLE slang_result)
 string(REGEX MATCH "([0-9]+\\.[0-9]+\\.[0-9]+)" slang_match "${slang_version}${slang_error}")
 if(NOT slang_result EQUAL 0 OR NOT slang_match OR CMAKE_MATCH_1 VERSION_LESS "2026.14.1")
  message(FATAL_ERROR "Slang 2026.14.1 or newer is required; tested pin is 2026.18.2.")
 endif()
 set(NOGRAPHICSAPI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(NOGRAPHICSAPI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 add_subdirectory(${BASIX_NOGRAPHICSAPI_SOURCE} ${CMAKE_BINARY_DIR}/nographicsapi)
 if(NOT EXISTS "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_REVISION")
  message(FATAL_ERROR "NoGraphicsAPI source has not been prepared with the Basix experimental interop patch.")
 endif()
 file(READ "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_REVISION" prepared_revision)
 string(STRIP "${prepared_revision}" prepared_revision)
 if(NOT prepared_revision STREQUAL "ffd9f2ab6b7442dcfbe0cb0051d84d0edb078d89")
  message(FATAL_ERROR "Prepared NoGraphicsAPI revision marker does not match the tested pin.")
 endif()
 if(NOT EXISTS "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_PATCH_VERSION")
  message(FATAL_ERROR "Re-prepare NoGraphicsAPI with current texture-interop patch version 2.")
 endif()
 file(READ "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_PATCH_VERSION" patch_version)
 string(STRIP "${patch_version}" patch_version)
 if(NOT patch_version STREQUAL "2")
  message(FATAL_ERROR "Unsupported Basix interop patch version.")
 endif()
 foreach(shader IN ITEMS pointer_smoke ray_query texture_smoke native_texture)
  set(extra_shader_capabilities)
  if(shader STREQUAL "ray_query")
   list(APPEND extra_shader_capabilities -capability spvRayQueryKHR)
  endif()
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/shaders/${shader}.spv
   COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/shaders
   COMMAND ${BASIX_SLANGC} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.slang -entry computeMain -stage compute -target spirv -profile spirv_1_5 -emit-spirv-directly -fvk-use-entrypoint-name -fvk-use-c-layout -matrix-layout-row-major -capability spvDescriptorHeapEXT ${extra_shader_capabilities} -I ${BASIX_NOGRAPHICSAPI_SOURCE}/include -o ${CMAKE_BINARY_DIR}/shaders/${shader}.spv
   DEPENDS shaders/${shader}.slang ${BASIX_NOGRAPHICSAPI_SOURCE}/include/NoGraphicsAPI/shader.slang ${BASIX_SLANGC} VERBATIM)
 endforeach()
 add_executable(basix_gpu_spike src/gpu_spike.cpp ${CMAKE_BINARY_DIR}/shaders/pointer_smoke.spv ${CMAKE_BINARY_DIR}/shaders/ray_query.spv ${CMAKE_BINARY_DIR}/shaders/texture_smoke.spv ${CMAKE_BINARY_DIR}/shaders/native_texture.spv)
 target_link_libraries(basix_gpu_spike PRIVATE NoGraphicsAPI::NoGraphicsAPI Vulkan::Vulkan)
 target_compile_definitions(basix_gpu_spike PRIVATE BASIX_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders")
 if(BUILD_TESTING)
  add_test(NAME gpu_pointer_and_ray_query COMMAND basix_gpu_spike --output ${CMAKE_BINARY_DIR}/gpu-output)
  set_tests_properties(gpu_pointer_and_ray_query PROPERTIES TIMEOUT 60 FAIL_REGULAR_EXPRESSION "NoGraphicsAPI validation:|Basix validation:")
  option(BASIX_VALIDATION_TESTS "Require available Khronos validation layers in Debug GPU tests" OFF)
  if(BASIX_VALIDATION_TESTS)
   add_test(NAME gpu_validation_interop COMMAND basix_gpu_spike --validate --cycles 3 --output ${CMAKE_BINARY_DIR}/validation-output CONFIGURATIONS Debug)
   set_tests_properties(gpu_validation_interop PROPERTIES TIMEOUT 120 FAIL_REGULAR_EXPRESSION "NoGraphicsAPI validation:|Basix validation:")
  endif()
 endif()
endif()
