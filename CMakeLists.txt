 endif()
 if(NOT BASIX_NOGRAPHICSAPI_SOURCE)
  message(FATAL_ERROR "Set BASIX_NOGRAPHICSAPI_SOURCE to a prepared pinned copy. No dependency is downloaded automatically.")

 find_program(BASIX_SLANGC slangc REQUIRED)
 execute_process(COMMAND ${BASIX_SLANGC} -version OUTPUT_VARIABLE slang_version ERROR_VARIABLE slang_error RESULT_VARIABLE slang_result)
 string(REGEX MATCH "([0-9]+\\.[0-9]+\\.[0-9]+)" slang_match "${slang_version}${slang_error}")
 if(NOT slang_result EQUAL 0 OR NOT slang_match OR CMAKE_MATCH_1 VERSION_LESS "2026.14.1")
  message(FATAL_ERROR "Slang 2026.14.1 or newer is required; tested pin is 2026.18.2.")
 endif()
 set(NOGRAPHICSAPI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(NOGRAPHICSAPI_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 add_subdirectory(${BASIX_NOGRAPHICSAPI_SOURCE} ${CMAKE_BINARY_DIR}/nographicsapi)
 if(NOT EXISTS "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_REVISION")
  message(FATAL_ERROR "NoGraphicsAPI source has not been prepared with the Basix experimental interop patch.")
 endif()
 file(READ "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_REVISION" prepared_revision)
 string(STRIP "${prepared_revision}" prepared_revision)
 if(NOT prepared_revision STREQUAL "ffd9f2ab6b7442dcfbe0cb0051d84d0edb078d89")
  message(FATAL_ERROR "Prepared NoGraphicsAPI revision marker does not match the tested pin.")
 endif()
 if(NOT EXISTS "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_PATCH_VERSION")
  message(FATAL_ERROR "Re-prepare NoGraphicsAPI with current interop patch version 6.")
 endif()
 file(READ "${BASIX_NOGRAPHICSAPI_SOURCE}/BASIX_INTEROP_PATCH_VERSION" patch_version)
 string(STRIP "${patch_version}" patch_version)
 if(NOT patch_version STREQUAL "6")
  message(FATAL_ERROR "Unsupported Basix interop patch version.")
 endif()
 foreach(shader IN ITEMS pointer_smoke ray_query texture_smoke native_texture material_conformance material_sampling material_visible material_distribution progressive_pt)
  set(extra_shader_capabilities)
  if(shader STREQUAL "ray_query" OR shader STREQUAL "progressive_pt")
   list(APPEND extra_shader_capabilities -capability spvRayQueryKHR)
  endif()
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/shaders/${shader}.spv
   COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/shaders
   COMMAND ${BASIX_SLANGC} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${shader}.slang -entry computeMain -stage compute -target spirv -profile spirv_1_5 -emit-spirv-directly -fvk-use-entrypoint-name -fvk-use-c-layout -matrix-layout-row-major -fp-mode precise -capability spvDescriptorHeapEXT ${extra_shader_capabilities} -I ${BASIX_NOGRAPHICSAPI_SOURCE}/include -o ${CMAKE_BINARY_DIR}/shaders/${shader}.spv
   DEPENDS shaders/${shader}.slang shaders/basix_material.slang shaders/basix_material_vndf.slang shaders/basix_ray_spawn.slang ${BASIX_NOGRAPHICSAPI_SOURCE}/include/NoGraphicsAPI/shader.slang ${BASIX_SLANGC} VERBATIM)
 endforeach()

 if(BASIX_BUILD_DLSS_SPIKE)
  if(NOT WIN32 OR NOT EXISTS "${BASIX_STREAMLINE_SDK}/include/sl_dlss.h")
   message(FATAL_ERROR "DLSS spike needs Windows and explicitly supplied Streamline SDK headers.")
  endif()
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/dependencies.lock.json" basix_dependency_lock)
  foreach(header IN ITEMS sl.h sl_version.h sl_core_types.h sl_dlss.h sl_helpers_vk.h)
   string(JSON expected_header_sha GET "${basix_dependency_lock}" dependencies Streamline-SR header_sha256 "${header}")
   file(SHA256 "${BASIX_STREAMLINE_SDK}/include/${header}" actual_header_sha)
   if(NOT actual_header_sha STREQUAL expected_header_sha)
    message(FATAL_ERROR "Streamline header differs from the explicit tested pin: ${header}")
   endif()
  endforeach()
  add_custom_command(OUTPUT ${CMAKE_BINARY_DIR}/shaders/dlss_readback.spv
   COMMAND ${BASIX_SLANGC} ${CMAKE_CURRENT_SOURCE_DIR}/shaders/dlss_readback.slang -entry computeMain -stage compute -target spirv -profile spirv_1_5 -emit-spirv-directly -fvk-use-entrypoint-name -fvk-use-c-layout -matrix-layout-row-major -capability spvDescriptorHeapEXT -I ${BASIX_NOGRAPHICSAPI_SOURCE}/include -o ${CMAKE_BINARY_DIR}/shaders/dlss_readback.spv
   DEPENDS shaders/dlss_readback.slang ${BASIX_NOGRAPHICSAPI_SOURCE}/include/NoGraphicsAPI/shader.slang ${BASIX_SLANGC} VERBATIM)
  add_executable(basix_dlss_spike src/dlss_spike.cpp src/streamline_runtime.cpp ${CMAKE_BINARY_DIR}/shaders/dlss_readback.spv)
  target_include_directories(basix_dlss_spike PRIVATE include ${BASIX_STREAMLINE_SDK}/include)
  target_link_libraries(basix_dlss_spike PRIVATE NoGraphicsAPI::NoGraphicsAPI Vulkan::Vulkan wintrust crypt32)
  target_compile_definitions(basix_dlss_spike PRIVATE BASIX_SHADER_DIR="${CMAKE_BINARY_DIR}/shaders")
